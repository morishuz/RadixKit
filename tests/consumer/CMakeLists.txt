cmake_minimum_required(VERSION 3.20)
project(radixkit_consumer LANGUAGES CXX)
if(RADIXKIT_SOURCE_DIR)
  add_subdirectory("${RADIXKIT_SOURCE_DIR}" radixkit-build)
  if(TARGET radixkit_sort_test OR TARGET radixkit_example OR TARGET radixkit_sort_bench OR TARGET radixkit_selection_bench)
    message(FATAL_ERROR "Embedding radixkit unexpectedly builds development targets")
  endif()
else()
  find_package(radixkit 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE radixkit::radixkit)
enable_testing()
add_test(NAME radixkit_consumer COMMAND consumer)
